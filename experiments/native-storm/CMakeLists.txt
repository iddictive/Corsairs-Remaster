cmake_minimum_required(VERSION 3.24)
project(storm_native_probe LANGUAGES C CXX)
option(STORM_ENABLE_AUDIO "Build the FMOD sound service (requires native FMOD)" ON)
if(NOT STORM_ENABLE_AUDIO)
  add_compile_definitions(STORM_AUDIO_DISABLED=1)
endif()

# Bring-up against an immutable engine snapshot; never writes into the playable runtime.
set(STORM_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.cache/storm" CACHE PATH "Engine snapshot")
set(CONAN_DATA "$ENV{HOME}/Library/Application Support/CorsairsBuild/conan-home/.conan/data" CACHE PATH "Existing dependency sources")
set(D3D9_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.cache/d3d9/dxvk-native" CACHE PATH "Native D3D9 headers")
set(NATIVE_D3D9_LIBS "dxvk_d3d9" CACHE STRING "Native D3D9 library")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(SDL2 REQUIRED)
find_package(ZLIB REQUIRED)
set(SDL2_LIBRARIES SDL2::SDL2)
add_library(zlib ALIAS ZLIB::ZLIB)
add_compile_options(-fsigned-char)
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/.cache")
add_compile_definitions(SSE2NEON_PRECISE_MINMAX=1 SSE2NEON_PRECISE_SQRT=1 SSE2NEON_PRECISE_DIV=1)
file(GLOB fmt_headers "${CONAN_DATA}/fmt/8.0.1/_/_/package/*/include")
include_directories(SYSTEM ${fmt_headers})
include_directories(SYSTEM /opt/homebrew/include)
add_compile_definitions(SPDLOG_FMT_EXTERNAL FMT_HEADER_ONLY)
include_directories("${D3D9_SOURCE}/include/native/directx" "${D3D9_SOURCE}/include/native/windows")

foreach(dep IN ITEMS directx fast_float spdlog fmod)
  add_library(${dep} INTERFACE)
endforeach()
file(GLOB ff_headers "${CONAN_DATA}/fast_float/3.4.0/_/_/package/*/include")
file(GLOB spd_headers "${CONAN_DATA}/spdlog/1.9.2/_/_/package/*/include")
file(GLOB fmod_headers "${CONAN_DATA}/fmod/2.02.05/piratesahoy+storm-engine/stable/package/*/include")
target_include_directories(fast_float INTERFACE ${ff_headers})
target_include_directories(spdlog INTERFACE ${spd_headers})
target_include_directories(fmod INTERFACE ${fmod_headers})
set(FMOD_LIBRARY "fmod" CACHE STRING "Native FMOD library")
target_link_libraries(fmod INTERFACE "${FMOD_LIBRARY}")

set(MI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(MI_OVERRIDE OFF CACHE BOOL "" FORCE)
set(MI_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(MI_BUILD_OBJECT OFF CACHE BOOL "" FORCE)
add_subdirectory("${CONAN_DATA}/mimalloc/2.1.7/_/_/source/src" mimalloc)
add_library(mimalloc ALIAS mimalloc-static)
set(SENTRY_BACKEND none CACHE STRING "" FORCE)
set(SENTRY_TRANSPORT none CACHE STRING "" FORCE)
set(SENTRY_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SENTRY_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SENTRY_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${CONAN_DATA}/sentry-native/0.6.5/_/_/source/src" sentry)
add_library(sentry-native ALIAS sentry)

include("${STORM_SOURCE}/cmake/StormSetup.cmake")
set(SRC_DIRS src)
set(PUBLIC_INCLUDE_DIRS include)
set(TESTSUITE_DIRS "")
set(RESOURCE_DIRS rsrc)
set(SOURCE_FILE_EXTENSIONS "*.cpp")
set(HEADER_FILE_EXTENSIONS "*.h" "*.hpp")
set(STORM_WATERMARK_FILE "${CMAKE_BINARY_DIR}/watermark.hpp")
file(WRITE "${STORM_WATERMARK_FILE}" "#define STORM_BUILD_WATERMARK native-probe-4860fe13\n")
set(global_modules_list "" CACHE INTERNAL "Active native modules" FORCE)
set(STORM_NATIVE_AUDIO_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/audio")
add_subdirectory("${STORM_SOURCE}/src" storm)
target_sources(renderer PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/raw_texture.cpp")
target_include_directories(renderer PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")

add_executable(extern_function_expression tests/extern_function_expression.cpp)
target_include_directories(extern_function_expression PRIVATE
  "${STORM_SOURCE}/src/libs/core/src"
)
target_link_libraries(extern_function_expression PRIVATE core)

# Apple ld preserves static module registration with force_load, not GNU whole-archive.
set_property(TARGET engine PROPERTY LINK_FLAGS "")
foreach(module IN LISTS global_modules_list)
  if(TARGET ${module})
    target_link_options(engine PRIVATE "LINKER:-force_load,$<TARGET_FILE:${module}>")
  endif()
endforeach()
