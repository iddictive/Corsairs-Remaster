foreach(name IN ITEMS perlworlnoise.tga worlnoise.bmp weather.bmp)
  string(JSON expected_hash GET "${sky_noise_manifest}" files "${name}")
  file(SHA256 "${SKY_NOISE_DIR}/${name}" actual_hash)
  if(NOT actual_hash STREQUAL expected_hash)
    message(FATAL_ERROR "Sky noise input differs from its source manifest: ${name}")
  endif()
  list(APPEND sky_noise_inputs "${SKY_NOISE_DIR}/${name}")
endforeach()
