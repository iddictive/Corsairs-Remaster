cmake_minimum_required(VERSION 3.24)
project(storm_metal_candidate LANGUAGES C CXX OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD 20)
set(NATIVE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../native-storm")
set(STORM_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.cache/storm" CACHE PATH "Isolated engine snapshot")
set(D3D9_SOURCE "${NATIVE_ROOT}/.cache/d3d9/dxvk-native" CACHE PATH "D3D9 interface headers only")
find_package(SDL2 REQUIRED)
add_library(storm_metal STATIC backend.mm dev_telemetry.cpp)
target_compile_options(storm_metal PRIVATE -fobjc-arc)
target_include_directories(storm_metal PUBLIC
  "${D3D9_SOURCE}/include/native/directx"
  "${D3D9_SOURCE}/include/native/windows")
target_link_libraries(storm_metal PUBLIC SDL2::SDL2
  "-framework Metal" "-framework Cocoa" "-framework QuartzCore")
set(NATIVE_D3D9_LIBS storm_metal CACHE STRING "Direct Metal backend" FORCE)
add_compile_definitions(STORM_METAL_DYNAMIC_LIGHTING_BRIDGE=1 STORM_METAL_ISLAND_GEOMETRY_BRIDGE=1 STORM_METAL_SCENE_HUD_BOUNDARY=1)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.cache/native-project" native-storm)
target_include_directories(geometry PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(metal-probe probe.cpp)
target_sources(metal-probe PRIVATE "${STORM_SOURCE}/src/libs/renderer/src/storm_d3dx9.cpp")
target_include_directories(metal-probe PRIVATE "${STORM_SOURCE}/src/libs/util/include")
target_link_libraries(metal-probe PRIVATE storm_metal)
set_target_properties(metal-probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_executable(sea-probe sea_probe.cpp)
target_link_libraries(sea-probe PRIVATE storm_metal)
set_target_properties(sea-probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_executable(particles-caustic-probe particles_caustic_probe.cpp)
target_link_libraries(particles-caustic-probe PRIVATE storm_metal)
set_target_properties(particles-caustic-probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(probe IN ITEMS land_shader postprocess stars ffp_generated depth_water water_camera lighting outdoor_lighting shadow weapon indexed dynamic_lighting character_contact ffp_frame_cache indexed_span_frame_cache progressive_shadow_softness cinematic_postfx scene_hud_boundary)
  add_executable(${probe}-probe ${probe}_probe.cpp)
  target_link_libraries(${probe}-probe PRIVATE storm_metal)
  set_target_properties(${probe}-probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_executable(cinematic-probe cinematic_probe.cpp)
set_source_files_properties(cinematic_probe.cpp PROPERTIES LANGUAGE OBJCXX)
target_compile_options(cinematic-probe PRIVATE -fobjc-arc)
target_link_libraries(cinematic-probe PRIVATE "-framework Metal" "-framework Foundation")
set_target_properties(cinematic-probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_executable(frame-arena-probe frame_arena_probe.cpp)
set_source_files_properties(frame_arena_probe.cpp PROPERTIES LANGUAGE OBJCXX)
target_compile_options(frame-arena-probe PRIVATE -fobjc-arc)
target_link_libraries(frame-arena-probe PRIVATE "-framework Metal" "-framework Foundation")
set_target_properties(frame-arena-probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

add_executable(timing-probe timing_probe.cpp)
target_include_directories(timing-probe PRIVATE "${STORM_SOURCE}/src/libs/core/src")
set_target_properties(timing-probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

add_executable(island-pn-probe island_pn_probe.cpp)
set_target_properties(island-pn-probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

add_executable(dev-telemetry-probe dev_telemetry_probe.cpp dev_telemetry.cpp)
set_target_properties(dev-telemetry-probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
