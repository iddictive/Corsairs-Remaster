from pathlib import Path
import subprocess
import sys

root = Path(__file__).resolve().parents[3]
evidence = Path(__file__).resolve().parent
work = Path(sys.argv[1]).resolve()
runner = (evidence / 'native-consumer-runner.py.txt').read_text()
runner = runner.replace('root=Path(__file__).resolve().parents[3]', f'root=Path({str(root)!r})')
runner = runner.replace("(evidence/'native-consumer-body.cpp.txt').read_text()", "(work/'body.txt').read_text()")
runner = runner.replace('(evidence/filename).write_text(out)', '(work/filename).write_text(out)')
(work / 'runner.py').write_text(runner)
body = (evidence / 'native-consumer-body.cpp.txt').read_text()
repeat = '''{ const auto state=ship.State; TOUCH_PARAMS first{},second{}; auto t=std::chrono::steady_clock::now(); ship.TouchMove(500,nullptr,&first); auto t1=std::chrono::steady_clock::now(); ship.TouchMove(500,nullptr,&second); auto t2=std::chrono::steady_clock::now(); check(!memcmp(&state,&ship.State,sizeof(state)),"repeat_state_unchanged"); printf("repeat_prediction first_ms=%.6f second_ms=%.6f pos_delta=(%.9g,%.9g,%.9g) exact=%d\\n",std::chrono::duration<double,std::milli>(t1-t).count(),std::chrono::duration<double,std::milli>(t2-t1).count(),first.vPos.x-second.vPos.x,first.vPos.y-second.vPos.y,first.vPos.z-second.vPos.z,memcmp(&first.vPos,&second.vPos,sizeof(first.vPos))==0); check(!memcmp(&first.vPos,&second.vPos,sizeof(first.vPos)),"repeat_prediction_exact"); }'''
sequence = '''{ TOUCH_PARAMS ignored{},afterOne{},afterTwo{}; const auto state=ship.State; ship.TouchMove(500,nullptr,&ignored); ship.TouchMove(650,nullptr,&afterOne); ship.TouchMove(500,nullptr,&ignored); ship.TouchMove(500,nullptr,&ignored); ship.TouchMove(650,nullptr,&afterTwo); printf("subsequent exact=%d delta=(%.9g,%.9g,%.9g)\\n",memcmp(&afterOne.vPos,&afterTwo.vPos,sizeof(afterOne.vPos))==0,afterOne.vPos.x-afterTwo.vPos.x,afterOne.vPos.y-afterTwo.vPos.y,afterOne.vPos.z-afterTwo.vPos.z); check(!memcmp(&state,&ship.State,sizeof(state)),"sequence_state_unchanged"); check(!memcmp(&afterOne.vPos,&afterTwo.vPos,sizeof(afterOne.vPos)),"sequence_exact"); }'''
dry = 'const auto before=ship.State;TOUCH_PARAMS a{},b{};'
contact = 'ship.print("native_terrain_contact");'
assert body.count(dry) == body.count(contact) == 1
body = body.replace(dry, repeat + '\n' + dry)
body = body.replace(contact, contact + '\n' + sequence + '\n' + repeat)
(work / 'body.txt').write_text(body)
subprocess.run([sys.executable, str(work / 'runner.py'), str(work)], check=True)
