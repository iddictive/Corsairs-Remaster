#include <Jolt/Physics/Collision/Shape/ScaledShape.h>
#include <Jolt/Physics/Collision/RayCast.h>
#include <Jolt/Physics/Collision/CastResult.h>
static void normalizedHullProbe(const storm::ship::contact::MaterialSnapshot &material)
{
 roundhouse::ReactiveWorld registration;JPH::StaticCompoundShapeSettings compound;
 double volume=0,first[3]{},second[3][3]{},maxSupportError=0,maxRayError=0;size_t count=0;
 auto start=std::chrono::steady_clock::now();
 for(const auto &p:storm::ship::contact::materialRoot(material).pieces){
  double lo[3]={1e30,p.lowerY,1e30},hi[3]={-1e30,p.upperY,-1e30},c[3],s[3];
  for(auto v:p.hull.points){lo[0]=std::min(lo[0],v.x);hi[0]=std::max(hi[0],v.x);lo[2]=std::min(lo[2],v.z);hi[2]=std::max(hi[2],v.z);}
  for(int i=0;i<3;i++){c[i]=(lo[i]+hi[i])*.5;s[i]=std::max((hi[i]-lo[i])*.5,1e-6);}
  JPH::Array<JPH::Vec3> points;double cx=0,cz=0;
  for(auto v:p.hull.points){cx+=v.x;cz+=v.z;for(double y:{p.lowerY,p.upperY})points.emplace_back(float((v.x-c[0])/s[0]),float((y-c[1])/s[1]),float((v.z-c[2])/s[2]));}
  cx/=p.hull.points.size();cz/=p.hull.points.size();
  JPH::ConvexHullShapeSettings settings(points,0.f);settings.mHullTolerance=1e-7f;
  auto result=settings.Create();if(result.HasError()){printf("normalized_rejected=%zu error=%s\n",count,result.GetError().c_str());check(false,"normalized_all_pieces_admitted");}
  auto *convex=static_cast<JPH::ConvexShape*>(result.Get().GetPtr());const JPH::Vec3 scale{float(s[0]),float(s[1]),float(s[2])};
  JPH::RefConst<JPH::Shape> scaled=new JPH::ScaledShape(convex,scale);JPH::Vec3 translation{float(c[0]),float(c[1]),float(c[2])};
  compound.AddShape(translation,JPH::Quat::sIdentity(),scaled);
  JPH::ConvexShape::SupportBuffer buffer;auto *support=convex->GetSupportFunction(JPH::ConvexShape::ESupportMode::IncludeConvexRadius,buffer,scale);
  for(int x=-1;x<=1;x++)for(int y=-1;y<=1;y++)for(int z=-1;z<=1;z++)if(x||y||z){JPH::Vec3 dir{float(x),float(y),float(z)};auto point=support->GetSupport(dir)+scaled->GetCenterOfMass()+translation;double expected=-1e30;for(auto v:p.hull.points)for(double py:{p.lowerY,p.upperY})expected=std::max(expected,x*v.x+y*py+z*v.z);maxSupportError=std::max(maxSupportError,std::abs(point.Dot(dir)-expected));}
  const double h=p.upperY-p.lowerY;JPH::RayCast ray(JPH::Vec3(float(cx-c[0]),float(p.upperY+h-c[1]),float(cz-c[2]))-scaled->GetCenterOfMass(),JPH::Vec3(0.f,float(-3*h),0.f));JPH::RayCastResult hit;
  check(scaled->CastRay(ray,JPH::SubShapeIDCreator(),hit),"normalized_piece_ray");maxRayError=std::max(maxRayError,std::abs(double(hit.mFraction)-1./3.));
  // Independent uniform prism integrals, using the native double polygon.
  double a=0,x1=0,z1=0,x2=0,z2=0,xz=0;
  for(size_t i=0;i<p.hull.points.size();i++){auto u=p.hull.points[i],v=p.hull.points[(i+1)%p.hull.points.size()];double cross=u.x*v.z-v.x*u.z;a+=cross;x1+=(u.x+v.x)*cross;z1+=(u.z+v.z)*cross;x2+=(u.x*u.x+u.x*v.x+v.x*v.x)*cross;z2+=(u.z*u.z+u.z*v.z+v.z*v.z)*cross;xz+=(2*u.x*u.z+u.x*v.z+v.x*u.z+2*v.x*v.z)*cross;}
  double sign=a<0?-1:1;a*=sign*.5;x1*=sign/6.;z1*=sign/6.;x2*=sign/12.;z2*=sign/12.;xz*=sign/24.;double ym=(p.lowerY+p.upperY)*.5,y2=(p.lowerY*p.lowerY+p.lowerY*p.upperY+p.upperY*p.upperY)/3.;
  volume+=a*h;first[0]+=x1*h;first[1]+=a*h*ym;first[2]+=z1*h;second[0][0]+=x2*h;second[1][1]+=a*h*y2;second[2][2]+=z2*h;second[0][1]+=x1*h*ym;second[0][2]+=xz*h;second[1][2]+=z1*h*ym;count++;
 }
 auto result=compound.Create();check(!result.HasError(),"normalized_compound_admitted");auto props=result.Get()->GetMassProperties();auto center=result.Get()->GetCenterOfMass();double maxInertiaRelative=0,centerError=0;
 for(int i=0;i<3;i++){centerError=std::max(centerError,std::abs(double(center[i])-first[i]/volume));for(int j=i;j<3;j++){double oracle=0;if(i==j){for(int k=0;k<3;k++)if(k!=i)oracle+=second[k][k]-first[k]*first[k]/volume;}else oracle=-(second[i][j]-first[i]*first[j]/volume);maxInertiaRelative=std::max(maxInertiaRelative,std::abs(double(props.mInertia(i,j))-1000*oracle)/(1000*volume*100));}}
 printf("normalized_parts=%zu build_and_oracles_ms=%.4f max_support_error_m=%.9g max_ray_fraction_error=%.9g native_volume=%.12g mass_relative_error=%.9g center_error_m=%.9g inertia_error_per_mass_100m2=%.9g\n",count,std::chrono::duration<double,std::milli>(std::chrono::steady_clock::now()-start).count(),maxSupportError,maxRayError,volume,std::abs(props.mMass-1000*volume)/(1000*volume),centerError,maxInertiaRelative);
 auto timing=std::chrono::steady_clock::now();volatile float sink=0;for(int i=0;i<100;i++)sink+=result.Get()->GetMassProperties().mMass;printf("retained_hull_mass_properties_us=%.5f\n",std::chrono::duration<double,std::micro>(std::chrono::steady_clock::now()-timing).count()/100);
 timing=std::chrono::steady_clock::now();for(int i=0;i<100;i++){float total,submerged;JPH::Vec3 buoyancy;result.Get()->GetSubmergedVolume(JPH::Mat44::sTranslation(center),JPH::Vec3::sReplicate(1.f),JPH::Plane(JPH::Vec3::sAxisY(),1000.f),total,submerged,buoyancy);sink+=submerged;}printf("retained_hull_dry_submerged_us=%.5f sink=%.9g\n",std::chrono::duration<double,std::micro>(std::chrono::steady_clock::now()-timing).count()/100,double(sink));
 check(maxSupportError<1e-4&&maxRayError<1e-4&&std::abs(props.mMass-1000*volume)/(1000*volume)<1e-4&&centerError<1e-4&&maxInertiaRelative<1e-4,"normalized_affine_mass_oracles");
}
