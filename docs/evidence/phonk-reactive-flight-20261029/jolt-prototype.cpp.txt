#include <Jolt/Jolt.h>
#include <Jolt/RegisterTypes.h>
#include <Jolt/Core/Factory.h>
#include <Jolt/Core/JobSystemSingleThreaded.h>
#include <Jolt/Core/TempAllocator.h>
#include <Jolt/Physics/PhysicsSystem.h>
#include <Jolt/Physics/Body/BodyCreationSettings.h>
#include <Jolt/Physics/Collision/Shape/BoxShape.h>
#include <cstdio>
#include <chrono>
#include <memory>
#include <cmath>
using namespace JPH;
class Layers final:public BroadPhaseLayerInterface {public:uint GetNumBroadPhaseLayers()const override{return 1;}BroadPhaseLayer GetBroadPhaseLayer(ObjectLayer)const override{return BroadPhaseLayer(0);}};
class BroadFilter final:public ObjectVsBroadPhaseLayerFilter{public:bool ShouldCollide(ObjectLayer,BroadPhaseLayer)const override{return false;}};
class ObjectFilter final:public ObjectLayerPairFilter{public:bool ShouldCollide(ObjectLayer,ObjectLayer)const override{return false;}};
struct Run{
 Layers layers;BroadFilter broad;ObjectFilter object;TempAllocatorImpl allocator{1024*1024};JobSystemSingleThreaded jobs{cMaxPhysicsJobs};PhysicsSystem physics;BodyID id;
 Run(float mass,float pitch=0.f){physics.Init(4,0,8,8,layers,broad,object);physics.SetGravity(Vec3(0,-9.81f,0));
 BodyCreationSettings s(new BoxShape(Vec3(5,2,15)),RVec3(0,100,0),Quat::sRotation(Vec3::sAxisX(),pitch),EMotionType::Dynamic,0);
 s.mOverrideMassProperties=EOverrideMassProperties::CalculateInertia;s.mMassPropertiesOverride.mMass=mass;
 s.mLinearDamping=s.mAngularDamping=0;s.mAllowSleeping=false;s.mMaxLinearVelocity=10000;s.mMaxAngularVelocity=100;
 id=physics.GetBodyInterface().CreateAndAddBody(s,EActivation::Activate);}
 ~Run(){physics.GetBodyInterface().RemoveBody(id);physics.GetBodyInterface().DestroyBody(id);}
 void step(float dt,float thrust,float pitch,float roll){
 auto &b=physics.GetBodyInterface();Quat q=b.GetRotation(id);Vec3 force(0,0,0),torque(0,0,0);
 for(int i=0;i<4;i++){Vec3 r((i&1)?3.f:-3.f,-2.f,(i&2)?9.f:-9.f);float k=std::clamp(1.f+pitch*((i&2)?-1.f:1.f)+roll*((i&1)?1.f:-1.f),0.f,2.f);Vec3 f(0,thrust*.25f*k,0);force+=q*f;torque+=q*r.Cross(f);}
 b.AddForceAndTorque(id,force,torque,EActivation::Activate);
 physics.Update(dt,1,&allocator,&jobs);
 }
 void print(const char *name){auto&b=physics.GetBodyInterface();auto p=b.GetPosition(id);auto v=b.GetLinearVelocity(id);auto w=b.GetAngularVelocity(id);printf("%s p=(%.8f,%.8f,%.8f) v=(%.8f,%.8f,%.8f) w=(%.8f,%.8f,%.8f)\n",name,float(p.GetX()),float(p.GetY()),float(p.GetZ()),v.GetX(),v.GetY(),v.GetZ(),w.GetX(),w.GetY(),w.GetZ());}
};
int main(){RegisterDefaultAllocator();Factory::sInstance=new Factory;RegisterTypes();
 {Run r(1000);for(int i=0;i<240;i++)r.step(1.f/240,9810,0,0);r.print("hover");}
 {Run r(2000);for(int i=0;i<240;i++)r.step(1.f/240,9810,0,0);r.print("same_force_double_mass");}
 {Run r(1000);for(int i=0;i<240;i++)r.step(1.f/240,14715,0,0);r.print("one_second_thrust");for(int i=0;i<480;i++)r.step(1.f/240,0,0,0);r.print("two_seconds_release");}
 {Run r(1000,.5f);for(int i=0;i<240;i++)r.step(1.f/240,14715,0,0);r.print("tilted_force");}
 for(int hz:{30,60,120}){Run r(1000);for(int frame=0;frame<hz*3;frame++)for(int k=0;k<240/hz;k++)r.step(1.f/240,14715,.8f,.6f);char name[64];snprintf(name,sizeof(name),"wasd_%dhz",hz);r.print(name);}
 {Run r(1000);auto start=std::chrono::steady_clock::now();for(int i=0;i<2000;i++)r.step(1.f/240,9810,0,0);double us=std::chrono::duration<double,std::micro>(std::chrono::steady_clock::now()-start).count()/2000.;printf("jolt_full_step_us=%.4f\n",us);}
 UnregisterTypes();delete Factory::sInstance;Factory::sInstance=nullptr;
}
