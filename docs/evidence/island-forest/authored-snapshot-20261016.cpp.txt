using entid_t=uint64_t;
struct Entity{virtual ~Entity()=default;};struct MODEL:Entity{CMatrix mtx;};
struct FakeCore{std::map<entid_t,Entity*> models;Entity*GetEntityPointer(entid_t id){return models[id];}}core;
struct ISLAND_BASE{struct WorldSurface{entid_t model;CMatrix authored;};};
struct ISLAND:ISLAND_BASE{entid_t model_id=1,seabed_id=2,worldReflection=4;CMatrix mIslandOld,mSeaBedOld;std::vector<entid_t>WorldModels()const{return {1,2,3,4};}CMatrix WorldAuthoredMatrix(entid_t)const;std::vector<WorldSurface>GetWorldSurfaces()const;};
CMatrix ISLAND::WorldAuthoredMatrix(entid_t id) const
{
    if(id==model_id)return mIslandOld;
    if(id==seabed_id)return mSeaBedOld;
    auto*model=dynamic_cast<MODEL*>(core.GetEntityPointer(id));
    return model ? model->mtx : CMatrix();
}

std::vector<ISLAND_BASE::WorldSurface> ISLAND::GetWorldSurfaces() const
{
    std::vector<WorldSurface> result;
#ifdef STORM_METAL_SHIP_HULL_DAMAGE
    for(const auto id:WorldModels())if(id!=worldReflection)result.push_back({id,WorldAuthoredMatrix(id)});
#else
    if(model_id)result.push_back({model_id,WorldAuthoredMatrix(model_id)});
#endif
    return result;
}
