#include "island_forest.hpp"
#include <fstream>
#include <iostream>
int main(int argc,char**argv){
    if(argc!=2)return 1;
    using namespace storm_island_forest;
    for(unsigned variant=0;variant<VariantCount;variant++)for(unsigned lod=0;lod<LodCount;lod++){
        const auto mesh=makeMesh(variant,lod);
        const std::string prefix=std::string(argv[1])+"-"+std::to_string(variant)+"-"+std::to_string(lod);
        for(const auto&part:{std::pair{"branches",&mesh.branches},std::pair{"leaves",&mesh.leaves}}){
            std::ofstream output(prefix+"-"+part.first+".vertices",std::ios::binary);
            output.write(reinterpret_cast<const char*>(part.second->data()),part.second->size()*sizeof(Vertex));
        }
        std::cout<<prefix<<" branches="<<mesh.branches.size()<<" leaves="<<mesh.leaves.size()<<'\n';
    }
}
