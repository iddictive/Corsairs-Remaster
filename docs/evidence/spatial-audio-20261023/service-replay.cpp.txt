#include "core_impl.h"
#include <array>
#include <chrono>
#include <string>
#include <unordered_map>
#include <vector>
#define private public
#include "sound_service.h"
#undef private
#include <cstdio>
#include <filesystem>

int main(int argc, char **argv)
{
    if (argc < 2 || argc > 3) return 64;
    std::filesystem::current_path(argv[1]);
    core_internal.Init();
    core_internal.InitBase();
    auto service = std::make_unique<SoundService>();
    auto config = ma_engine_config_init();
    config.noDevice = MA_TRUE; config.channels = 2; config.sampleRate = 48000;
    if (ma_engine_init(&config, &service->engine_) != MA_SUCCESS) return 2;
    service->initialized_ = true;
    service->InitAliases();
    service->SetMasterVolume(.5f, .5f, .5f);
    service->SetCameraPosition(CVECTOR(0.f, 0.f, 0.f));
    auto clear = [&] { service->SoundStop(TSD_ID(SOUND_INVALID_ID), 0); };
    auto play = [&](const char *alias, float x, float z) {
        CVECTOR p(x, 0, z);
        return service->SoundPlay(alias, PCM_3D, VOLUME_FX, false, false, false, 0, &p, -1, -1, 0, 1, 128);
    };
    auto valid = [](TSD_ID id) { return int32_t(id) != SOUND_INVALID_ID; };
    auto count = [&] { int n=0; for (auto &s:service->slots_) n+=s.initialized; return n; };
    auto mix = [&] {
        std::vector<float> output(48000 * 2);
        ma_engine_read_pcm_frames(&service->engine_, output.data(), 48000, nullptr);
        double energy=0, peak=0; int clipped=0;
        for (float s:output) { energy+=double(s)*s; peak=std::max(peak,double(std::abs(s))); clipped+=std::abs(s)>1; }
        std::printf("\"peak\":%.9g,\"rms\":%.9g,\"clipped_samples\":%d",peak,std::sqrt(energy/output.size()),clipped);
    };
    std::srand(1);
    if (argc == 3 && std::string(argv[2]) == "cannon")
    {
        auto active = [&] { return service->ActiveVoiceCount(storm::metal::audio::VoiceClass::cannon); };
        for (int i=0; i<12; ++i) play("cannon_fire_cannon_16", float(i)*.5f, -10);
        const auto otherShip = play("cannon_fire_cannon_16", 0, -100);
        const auto ownNext = play("cannon_fire_cannon_16", 1, -10);
        std::printf("{\"case\":\"own_salvo_then_other_ship\",\"other_admitted\":%s,\"next_own_admitted\":%s,\"other_still_playing\":%s,\"active\":%zu,\"allocated\":%d}\n",
            valid(otherShip)?"true":"false",valid(ownNext)?"true":"false",
            service->SoundIsPlaying(otherShip)?"true":"false",active(),count());
        const auto beforeFailure = active();
        const bool failed = !valid(play("cannon_fire_missing", 0, -10));
        std::printf("{\"case\":\"failed_decode_preserves_voices\",\"failed\":%s,\"before\":%zu,\"after\":%zu}\n",
            failed?"true":"false",beforeFailure,active());
        clear();
        const auto started = std::chrono::steady_clock::now();
        int admitted = 0, maxAllocated = 0;
        for (int i=0; i<96; ++i) {
            admitted += valid(play("cannon_fire_cannon_16", float(i%12), -10));
            maxAllocated = std::max(maxAllocated, count());
        }
        const auto us = std::chrono::duration_cast<std::chrono::microseconds>(std::chrono::steady_clock::now()-started).count();
        std::printf("{\"case\":\"cannon_burst\",\"admitted\":%d,\"active\":%zu,\"max_allocated\":%d,\"play_us\":%lld,",
            admitted,active(),maxAllocated,(long long)us); mix(); std::puts("}");
        clear();
        admitted = 0; maxAllocated = 0;
        for (int i=0; i<32; ++i) {
            admitted += valid(play("cannon_fire_cannon_48", float(i%8)*35, -float(i/8)*35-10));
            maxAllocated = std::max(maxAllocated, count());
        }
        std::printf("{\"case\":\"distinct_cannon_sources\",\"admitted\":%d,\"active\":%zu,\"max_allocated\":%d,",
            admitted,active(),maxAllocated); mix(); std::puts("}");
        clear();
        CVECTOR cachedPosition(0.f, 0.f, -10.f);
        for (int i=0; i<12; ++i)
            service->SoundPlay("cannon_fire_cannon_16", PCM_3D, VOLUME_FX, true, false, false,
                0, &cachedPosition, -1, -1, 0, 1, 128);
        const bool cachedProtected = !valid(play("cannon_fire_cannon_16", 0, -100));
        const auto ui = service->SoundPlay("Button", PCM_STEREO, VOLUME_FX, false, false, false,
            0, nullptr, -1, -1, 0, 1, 128);
        std::printf("{\"case\":\"retained_cannon_controls\",\"cached_protected\":%s,\"cannons\":%zu,\"ui_admitted\":%s}\n",
            cachedProtected?"true":"false",active(),valid(ui)?"true":"false");
        clear();
        for (const char *alias : {"cannon_fire", "fort_cannon_fire", "cannon_fire_cannon_3",
             "cannon_fire_cannon_6", "cannon_fire_cannon_8", "cannon_fire_cannon_12",
             "cannon_fire_cannon_16", "cannon_fire_cannon_20", "cannon_fire_cannon_24",
             "cannon_fire_cannon_32", "cannon_fire_cannon_36", "cannon_fire_cannon_42",
             "cannon_fire_cannon_48", "cannon_fire_culverine_8", "cannon_fire_culverine_12",
             "cannon_fire_culverine_16", "cannon_fire_culverine_24", "cannon_fire_culverine_32"})
        {
            for (float distance : {10.f, 150.f, 250.f, 300.f, 500.f, 1000.f, 5000.f}) {
                std::srand(1);
                const auto shot = play(alias, 0, -distance);
                std::printf("{\"case\":\"cannon_distance\",\"alias\":\"%s\",\"distance\":%.0f,\"admitted\":%s,",
                    alias,distance,valid(shot)?"true":"false"); mix(); std::puts("}"); clear();
            }
        }
        return 0;
    }
    if (argc == 3 && std::string(argv[2]) == "distinct")
    {
        auto first = play("ball_splash", 0, -10);
        const float before = ma_sound_get_volume(&service->Find(first)->sound);
        int admitted = 0;
        for (const char *alias : {"fort_cann_explode", "ball2bort", "mast_fall"})
            admitted += valid(play(alias, 0, -10));
        std::printf("{\"case\":\"independent_effects\",\"other_admitted\":%d,\"before\":%.9g,\"after\":%.9g}\n",
            admitted, before, ma_sound_get_volume(&service->Find(first)->sound));
        clear();
        first = play("fort_cann_explode", 0, -10);
        const bool immediate = valid(play("fort_cann_explode", 0, -10));
        service->Find(first)->impactStarted -= std::chrono::milliseconds(70);
        const auto second = play("fort_cann_explode", 0, -10);
        const bool later = valid(second);
        for (auto &slot : service->slots_) if (slot.initialized)
            slot.impactStarted -= std::chrono::milliseconds(200);
        const bool third = valid(play("fort_cann_explode", 0, -10));
        const int repeatSlots = count();
        service->SoundRelease(second);
        const bool restored = ma_sound_get_volume(&service->Find(first)->sound) ==
            service->Find(first)->volume * .5f;
        std::printf("{\"case\":\"local_repeat\",\"immediate\":%s,\"after_70ms\":%s,\"third\":%s,\"slots\":%d,\"gain_restored\":%s}\n",
            immediate?"true":"false", later?"true":"false", third?"true":"false", repeatSlots, restored?"true":"false");
        clear();
        first = play("ball_splash", 0, -10);
        const float isolated = ma_sound_get_volume(&service->Find(first)->sound);
        play("ball_splash", 80, -10);
        std::printf("{\"case\":\"separate_regions\",\"before\":%.9g,\"after\":%.9g}\n",
            isolated, ma_sound_get_volume(&service->Find(first)->sound));
        clear();
        for (int i=0; i<8; ++i) play("ball_splash", float(i)*22, -10);
        const bool independent = valid(play("fort_cann_explode", 0, -10));
        const bool ninth = valid(play("ball_splash", 0, -60));
        std::printf("{\"case\":\"saturated_splash_group\",\"different_admitted\":%s,\"ninth_splash\":%s,\"slots\":%d}\n",
            independent?"true":"false", ninth?"true":"false", count());
        clear();
        admitted = 0;
        for (const char *alias : {"ball_splash", "fort_cann_explode", "ball2bort", "ball2sail",
             "mast_fall", "coll_ship2rock", "ship_explosion", "episode_boom", "fly_ball"})
            admitted += valid(play(alias, 0, -10));
        std::printf("{\"case\":\"mixed_battle_effects\",\"admitted\":%d,", admitted);
        mix(); std::puts("}");
        clear();
        return 0;
    }
    int admitted=0;
    auto started=std::chrono::steady_clock::now();
    for (int i=0;i<96;++i) admitted+=valid(play("ball_splash", float(i%12)*25, -float(i/12)*25-10));
    auto us=std::chrono::duration_cast<std::chrono::microseconds>(std::chrono::steady_clock::now()-started).count();
    std::printf("{\"case\":\"distributed_splashes\",\"admitted\":%d,\"slots\":%d,\"play_us\":%lld,",admitted,count(),(long long)us); mix(); std::puts("}");
    if (argc == 3 && std::string(argv[2]) == "density") { clear(); return 0; }
    const bool capped = !valid(play("ball2bort", 0, -10));
    for (auto &slot : service->slots_) if (slot.initialized) { service->Release(slot); break; }
    const bool replacement = valid(play("ball2bort", 0, -10));
    bool balanced = count() == 8;
    for (auto &slot : service->slots_) if (slot.initialized)
        balanced = balanced && slot.mixGain > 0.f && slot.mixGain <= 1.f;
    clear();
    const auto single = play("ball_splash", 0, -10);
    const bool restored = valid(single) && service->Find(single)->mixGain == 1.f;
    std::printf("{\"case\":\"impact_lifecycle\",\"mixed_alias_cap\":%s,\"replacement\":%s,\"balanced\":%s,\"single_gain_restored\":%s}\n",
        capped?"true":"false",replacement?"true":"false",balanced?"true":"false",restored?"true":"false");
    clear(); admitted=0;
    for (int i=0;i<64;++i) {
        for (auto &s:service->slots_) if (s.initialized) s.impactStarted-=std::chrono::milliseconds(100);
        admitted+=valid(play("fort_cann_explode",0,-10));
    }
    std::printf("{\"case\":\"local_destruction_100ms\",\"admitted\":%d,\"slots\":%d,",admitted,count()); mix(); std::puts("}");
    clear();
    for (int i=0;i<12;++i) play("fort_cannon_fire",float(i),-5000);
    bool near=valid(play("fort_cannon_fire",0,-20));
    std::printf("{\"case\":\"far_cannons_then_near\",\"near_admitted\":%s,\"slots\":%d}\n",near?"true":"false",count());
    clear();
    for (const char *alias:{"ball_splash","coll_ship2rock","fort_cann_explode","thunder"}) {
        for (float distance:{10.f,200.f,5000.f,10000.f}) {
            play(alias,0,-distance);
            std::printf("{\"case\":\"distance\",\"alias\":\"%s\",\"distance\":%.0f,",alias,distance);mix();std::puts("}");clear();
        }
    }
    // Stereo UI has no distance admission; loop owners must be able to move
    // an initially distant loop back into range without losing its handle.
    auto ui=service->SoundPlay("Button",PCM_STEREO,VOLUME_FX,false,false,false,0,nullptr,-1,-1,0,1,128);
    CVECTOR far(0.f,0.f,-5000.f);
    auto loop=service->SoundPlay("ship_onfire",PCM_3D,VOLUME_FX,false,true,false,0,&far,-1,-1,0,1,128);
    std::printf("{\"case\":\"retained_controls\",\"stereo\":%s,\"distant_loop\":%s}\n",valid(ui)?"true":"false",valid(loop)?"true":"false");
    clear();
}
