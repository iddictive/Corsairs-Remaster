#include "missile_trail_vfx.hpp"
#include <cstdio>
#include <cassert>
using namespace storm_metal;
int main(){
 MissileTrailVfx v; v.beginFrame(.016f);v.burst({0,0,0},MissileVfxType::Kalibr);
 for(int i=0;i<240;i++){v.beginFrame(1.f/60.f);v.emitSegment({0,20,0},{0,20,0},1.f/60.f,MissileVfxType::Kalibr);} assert(v.storedCount()==16);
 v.clear();for(int i=0;i<120;i++){v.beginFrame(1.f/60.f);v.emitSegment({float(i)/6.f,0,0},{float(i+1)/6.f,0,0},1.f/60.f,MissileVfxType::Kalibr);} auto slow=v.storedCount();assert(slow>=3&&slow<=6);
 v.clear();v.beginFrame(.016f);v.emitSegment({0,0,0},{100000,0,0},.016f,MissileVfxType::Kinzhal);assert(v.storedCount()==24);v.beginFrame(5.f);assert(v.storedCount()==0&&!v.hasVisible());
 puts("PASS stationary 4s: no trail puffs; 10m/s 2s: bounded moving emission; huge segment=24; gap reset; burst retained");printf("slow puffs %zu\n",slow);
}
