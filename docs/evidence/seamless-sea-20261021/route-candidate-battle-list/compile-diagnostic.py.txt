# CPU-only diagnostic of the same full PROGRAM projection; never runs Main.
# Existing script-vm-probe returns 65 for a failed prelude, or the deliberate
# duplicate Main in its subsequent driver. That latter marker proves the full
# prelude compiled while preventing Main execution.
import hashlib, json, os, pathlib, shutil, subprocess, sys
repo=pathlib.Path(os.environ['CORSAIRS_REPO_ROOT']).resolve()
scratch=pathlib.Path(os.environ['CODEX_SCRATCH_DIR'])
out=pathlib.Path(os.environ['CORSAIRS_EVIDENCE_OUTPUT']).resolve()
sys.path.insert(0,str(repo/'tools'))
from gameplay_sources import read
projected,inputs=read()
root=scratch/'PROGRAM'
shutil.copytree(repo/'experiments/native-metal/.cache/runtime/PROGRAM',root,symlinks=False)
for name,(raw,legacy,encoding) in projected.items():
    if name.startswith('PROGRAM/'):
        path=scratch/name;path.parent.mkdir(parents=True,exist_ok=True);path.write_bytes(raw)
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
tree=hashlib.sha256(''.join(str(p.relative_to(root))+':'+sha(p)+'\n' for p in sorted(root.rglob('*')) if p.is_file()).encode()).hexdigest()
(root/'engine.ini').write_text('[script]\ndebuginfo=1\ncodefiles=0\n')
(root/'cases.c').write_text('int CpuPreludePassedSentinel;\nint CpuPreludePassedSentinel;\nint DiagnosticNoop(){return 0;}\n')
binary=repo/'experiments/native-metal/.cache/build/bin/script-vm-probe'
# LoadFile binds storm-engine headers to executable-directory/resource/shared.
# Copy the existing binary only; no rebuild or canonical resource writes.
local_binary=scratch/'script-vm-probe';shutil.copyfile(binary,local_binary);local_binary.chmod(0o700)
shared=repo/'experiments/native-metal/engine/src/libs/shared_headers/include/shared'
(scratch/'RESOURCE').mkdir()
shutil.copytree(shared,scratch/'RESOURCE/shared',symlinks=False)
proc=subprocess.run([str(local_binary),str(root),'DiagnosticNoop','0','DiagnosticNoop','seadogs.c'],capture_output=True,text=True,timeout=10)
result={'kind':'CPU-only full PROGRAM compilation with source diagnostics','returncode':proc.returncode,'PROGRAM_tree_sha256':tree,'binary_sha256':sha(binary),'binary_owner':'experiments/native-metal/script_vm_probe.cpp','projection_owner':'tools/gameplay_sources.py:read() runtime_bytes','debug_setting':'[script] debuginfo=1','canonical_sources_unchanged':all(path.read_bytes()==raw for path,(raw,incoming) in inputs.items()),'stdout':proc.stdout,'stderr':proc.stderr}
result['script_shared_header_projection_owner']='experiments/native-metal/run.sh:126 canonical engine/src/libs/shared_headers/include/shared -> resource/shared'
result['script_shared_headers_sha256']={str(p.relative_to(shared)):sha(p) for p in sorted(shared.rglob('*')) if p.is_file()}
result['projected_program_files_sha256']={name:hashlib.sha256(row[0]).hexdigest() for name,row in projected.items() if name.startswith('PROGRAM/')}
result['prelude_passed']=False
for path in (root/'.vm-userdata/Logs').glob('*.log'):
    if path.name=='error.log' and ('Duplicate variable name: CpuPreludePassedSentinel' in path.read_text(errors='replace') or 'Function [Main] already declared in: seadogs.c line' in path.read_text(errors='replace')):
        result['prelude_passed']=True
    (out/('cpu-'+path.name)).write_text(path.read_text(errors='replace').replace(str(scratch),'<scratch>').replace(str(repo),'<repo>'))
(out/'cpu-compile.json').write_text(json.dumps(result,indent=2))
print(json.dumps({key:result[key] for key in ['kind','returncode','PROGRAM_tree_sha256','binary_sha256','canonical_sources_unchanged','prelude_passed']},indent=2))
