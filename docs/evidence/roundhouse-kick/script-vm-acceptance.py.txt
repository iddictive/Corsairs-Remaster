def extract(path, name):
    text = path.read_text(encoding='utf-8-sig')
    match = re.search(r'(?m)^(?:void|int|float|bool|string)\s+' + name +
                      r'\([^;\n]*\)[ \t]*(?://[^\n]*)?\s*\{', text)
    assert match, name
    start = text.index('{', match.start())
    masked = re.sub(r'//[^\n]*|/\*[\s\S]*?\*/|"(?:\\.|[^"\\])*"',
                    lambda m: ' ' * len(m[0]), text[start:])
    depth, end = 1, 1
    while depth:
        depth += (masked[end] == '{') - (masked[end] == '}')
        end += 1
    return text[match.start():start + end]


def run(directory, entry, prelude=None):
    args = [str(PROBE), str(directory), entry, '0']
    if prelude:
        args += [entry, prelude]
    completed = subprocess.run(args, cwd=directory, capture_output=True, text=True)
    error_file = directory / '.vm-userdata/Logs/error.log'
    result = dict(exit=completed.returncode, stdout=completed.stdout,
                  stderr=completed.stderr,

    print(json.dumps({directory.name: result}, indent=2))
    assert completed.returncode == 0 and not result['script_errors'], result
    return result
