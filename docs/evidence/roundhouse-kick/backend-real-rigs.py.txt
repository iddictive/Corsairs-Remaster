def full_mesh(source_mesh):
    path=here/"inputs"/Path(source_mesh["path"]).name
    raw=path.read_bytes()
    assert sha(path)==source_mesh["sha256"]
    vertices,_=animated_mesh(raw)
    assert json.loads(json.dumps(vertices))==source_mesh["vertices"]
    h=HEADER.unpack_from(raw)
    offset=HEADER.size+h[2]+4*h[3]+4*h[4]+MATERIAL.size*h[5]+80*h[6]+LABEL.size*h[7]+OBJECT.size*h[8]+h[9]*6
    descriptors=[struct.unpack_from("<2i",raw,offset+i*8)for i in range(h[10])]
    offset+=h[10]*8;normals=[]
    for typ,size in descriptors:
        assert typ==4 and size%44==0
        normals += [struct.unpack_from("<3f",raw,offset+44*i+20)for i in range(size//44)]
        offset+=size
    assert len(normals)==len(vertices)
    mesh_inputs.append(path)
    return [(point,weight,a,b,normal)for (point,weight,a,b),normal in zip(vertices,normals)]

# Use the existing GM parser for the visible payload offset, then the exact
# packed BSP layout consumed by GEOM (rdf.h/geom_static.cpp). Clip preserves
# triangle order and NODER::AddPolygon transforms each vertex without swapping.
sys.path.insert(0,str(repo / "tools"))
from patch_metal_portobello_ground import parse_gm
locations = ["Town_PortoBello/Town/PortoBello.gm", "Town_Caracas/Town/Caracas.gm", "Town_Havana/Town/Havana.gm"]
geometry_lines = [str(len(locations))]
geometry_report = []
def cross(a,b):
    return (a[1]*b[2]-a[2]*b[1],a[2]*b[0]-a[0]*b[2],a[0]*b[1]-a[1]*b[0])
for relative in locations:
    path = repo / "experiments/native-metal/.cache/runtime/RESOURCE/MODELS/locations" / relative
    raw = path.read_bytes()
    gm = parse_gm(raw)
    assert gm["header"][1] & 2
    start = gm["parsed_end"]
    nodes, vertices, triangles = struct.unpack_from("<3i", raw, start)
    vertex_start = start+12+nodes*24
    triangle_start = vertex_start+vertices*12
    assert triangle_start+triangles*9 == len(raw)
    points = [struct.unpack_from("<3f",raw,vertex_start+i*12) for i in range(vertices)]
    floor_candidates=[]; wall_candidates=[]; up=down=0
    for index in range(triangles):
        indices = [int.from_bytes(raw[triangle_start+index*9+j*3:triangle_start+index*9+j*3+3],"little") for j in range(3)]
        a,b,c = p = [points[j] for j in indices]
        normal = cross(tuple(b[j]-a[j] for j in range(3)),tuple(c[j]-a[j] for j in range(3)))
        area2 = math.sqrt(sum(v*v for v in normal))
        if area2 < 1: continue
        normal = tuple(v/area2 for v in normal)
        y = sum(v[1] for v in p)/3
        if abs(normal[1])>.99 and -.1<y<3:
            if normal[1]>0: up+=1
            else: down+=1
        if normal[1]>.999999 and -.1<y<3:
            floor_candidates.append((area2,index,p,normal))
        if abs(normal[1])<.00001 and max(v[1] for v in p)-min(v[1] for v in p)>3:
            wall_candidates.append((area2,index,p,normal))
    floor=max(floor_candidates); wall=max(wall_candidates)
    geometry_lines.append(f"{path.stem} {floor[1]} {wall[1]}")
    for p in floor[2]+wall[2]: geometry_lines.append(" ".join(format(x,".9g") for x in p))
    geometry_lines.append(" ".join(format(x,".9g") for x in wall[3]))
    geometry_report.append({"resource":relative,"sha256":sha(path),"bytes":len(raw),
        "bsp":{"nodes":nodes,"vertices":vertices,"triangles":triangles,"vertex_offset":vertex_start,"triangle_offset":triangle_start},
        "horizontal_ground_faces":{"up":up,"down":down},
        "floor":{"triangle":floor[1],"vertices":floor[2],"normal":floor[3]},
        "wall":{"triangle":wall[1],"vertices":wall[2],"normal":wall[3]}})
(scratch / "geometry.txt").write_text("\n".join(geometry_lines)+"\n")
