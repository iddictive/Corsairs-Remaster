def extract(text, name):
    match = re.search(r'(?m)^(?:void|int|float|bool|string)\s+' + name + r'\([^;\n]*\)\s*\{', text)
    assert match, name
    start = text.index('{', match.start())
    masked = re.sub(r'//[^\n]*|/\*[\s\S]*?\*/|"(?:\\.|[^"\\])*"', lambda m: ' ' * len(m[0]), text[start:])
    depth, end = 1, 1
    while depth:
        depth += (masked[end] == '{') - (masked[end] == '}')
        end += 1
    return text[match.start():start + end]
