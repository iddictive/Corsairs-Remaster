def extract(text, name):
    match = re.search(r'(?m)^(?:void|int|float|bool|string)\s+' + name + r'\([^;\n]*\)\s*\{', text)
    assert match, name
    start = text.index('{', match.start())
    masked = re.sub(r'//[^\n]*|/\*[\s\S]*?\*/|"(?:\\.|[^"\\])*"', lambda m: ' ' * len(m[0]), text[start:])
    depth, end = 1, 1
    while depth:
        depth += (masked[end] == '{') - (masked[end] == '}')
        end += 1
    return text[match.start():start + end]

def run(directory, prelude=None):
    args = [str(probe), str(directory), 'Run', '0']
    if prelude:
        args += ['Run', prelude]
    result = subprocess.run(args, capture_output=True, text=True, timeout=90)
    errors = directory / '.vm-userdata/Logs/error.log'
    record = {'exit': result.returncode, 'stdout': result.stdout, 'stderr': result.stderr,
              'script_errors': errors.read_text(errors='replace') if errors.exists() else ''}
    assert result.returncode == 0 and not record['script_errors'], record
    return record
