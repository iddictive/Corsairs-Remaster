struct Replay {
    Drift drift;
    float yaw=0, rate=0, course=0, previous=0, cap=.7f;
    void step(float turn,bool gas=true,bool brake=false,bool air=false,bool boost=false,float kick=0) {
        drift.advance(motorAcceleration,turn,yaw,dt,kick,drift.initialized?drift.lastForward:500.f,gas,brake,air,boost);
        const float next=std::atan2(drift.worldX,drift.worldZ);
        course+=std::remainder(next-previous,2*pi); previous=next;
        rate+=(yawSpeed(cap/yawBoost,turn)-rate)*-std::expm1(-5.f*dt);
        yaw+=rate*dt;
    }
    float slip() const {return std::remainder(yaw-previous,2*pi)*180/pi;}
};
