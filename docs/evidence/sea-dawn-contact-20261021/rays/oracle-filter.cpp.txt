#include "authored_sky.hpp"
#include "render_device_api.hpp"
#include "world_map_player_model.hpp"
#include <simd/simd.h>
#include <array>
#include <vector>
#include <fstream>
#include <cstdio>
#include <cmath>
#include <algorithm>
int main(int argc,char**argv){constexpr unsigned W=128,H=128,N=128;std::array<std::vector<uint32_t>,5>pixels;storm_render::Matrix identity{};identity._11=identity._22=identity._33=identity._44=1;
 for(unsigned f=0;f<5;++f){pixels[f].resize(N*N);for(unsigned y=0;y<N;++y)for(unsigned x=0;x<N;++x){unsigned r=30,g=80,b=180;if(f==4){r=20+200*x/(N-1);g=20+200*y/(N-1);b=200;}else if(y>N*9/10){r=g=b=((x/8+f)%2)?220:20;}pixels[f][y*N+x]=0xff000000|(r<<16)|(g<<8)|b;}}
 auto projection=identity;const float lens=std::tan(storm_map_player::perspective*.5f);projection._11=projection._22=1/lens;projection._33=100.f/99.f;projection._34=1;projection._43=-100.f/99.f;projection._44=0;
 auto shade=[&](simd_float3 ray,float height){simd_float3 origin={0,height,0};float distance=1.e10f;for(int axis=0;axis<3;++axis)if(std::abs(ray[axis])>1.e-7f){float bound=ray[axis]>0?1.f:-1.f;float t=(bound-origin[axis])/ray[axis];if(t>0)distance=std::min(distance,t);}auto point=origin+ray*distance;auto coordinate=storm_metal::authoredSkyCoordinate(point.x,std::abs(point.y),point.z);float u=(1.f/1024.f)+(1-2.f/1024.f)*coordinate.u,v=(1.f/1024.f)+((coordinate.face==4?1.f-1.f/1024.f:1.f)-1.f/1024.f)*coordinate.v;float fx=std::clamp(u*N-.5f,0.f,float(N-1)),fy=std::clamp(v*N-.5f,0.f,float(N-1));unsigned ix=unsigned(fx),iy=unsigned(fy);simd_float3 color{};for(int dy=0;dy<2;++dy)for(int dx=0;dx<2;++dx){auto packed=pixels[coordinate.face][std::min(iy+dy,N-1)*N+std::min(ix+dx,N-1)];float weight=(dx?fx-ix:1-(fx-ix))*(dy?fy-iy:1-(fy-iy));color+=simd_make_float3((packed>>16)&255,(packed>>8)&255,packed&255)*weight;}return color;
 };
 auto viewRay=[&](unsigned x,unsigned y,float pitch){
  simd_float4x4 pv{},vv{};memcpy(&pv,&projection,64);auto view=identity;view._22=view._33=std::cos(pitch);view._23=std::sin(pitch);view._32=-std::sin(pitch);memcpy(&vv,&view,64);const auto inv=simd_inverse(simd_mul(pv,vv));float nx=(x+.5f)/W*2-1,ny=1-(y+.5f)/H*2;auto a=simd_mul(inv,simd_make_float4(nx,ny,0,1)),b=simd_mul(inv,simd_make_float4(nx,ny,1,1));auto ray=simd_normalize(b.xyz/b.w-a.xyz/a.w);return ray;};

constexpr int FW=256,FH=128;std::vector<simd_float3>field(FW*FH);for(int y=0;y<FH;++y)for(int x=0;x<FW;++x){float phi=6.28318530718f*((x+.5f)/FW-.5f),l=(y+.5f)/FH*2-1,theta=l*l*(l<0?-1.f:1.f)*1.57079632679f;field[y*FW+x]=shade(simd_make_float3(std::cos(theta)*std::cos(phi),std::sin(theta),std::cos(theta)*std::sin(phi)),.6f);}
std::ifstream raw(argv[1],std::ios::binary);std::vector<uint32_t>image(W*H);raw.read((char*)image.data(),image.size()*4);double filtered=0,direct=0,resampling=0;float maxerr=0;unsigned n=0;
for(unsigned y=8;y<H-8;++y)for(unsigned x=8;x<W-8;++x){auto ray=viewRay(x,y,-storm_map_player::overheadPitch);float theta=std::asin(std::clamp(ray.y,-1.f,1.f));float u=std::atan2(ray.z,ray.x)/6.28318530718f+.5f,v=(theta<0?-1.f:1.f)*std::sqrt(std::abs(theta)/1.57079632679f)*.5f+.5f;float fx=u*FW-.5f,fy=v*FH-.5f;int ix=int(std::floor(fx)),iy=int(std::floor(fy));float tx=fx-ix,ty=fy-iy;simd_float3 ref{};for(int dy=0;dy<2;++dy)for(int dx=0;dx<2;++dx)ref+=field[std::clamp(iy+dy,0,FH-1)*FW+((ix+dx+FW)%FW)]*(dx?tx:1-tx)*(dy?ty:1-ty);auto src=shade(ray,.6f);auto p=image[y*W+x];auto actual=simd_make_float3((p>>16)&255,(p>>8)&255,p&255);auto e=actual-ref;filtered+=simd_dot(e,e);e=actual-src;direct+=simd_dot(e,e);e=ref-src;resampling+=simd_dot(e,e);maxerr=std::max(maxerr,simd_reduce_max(simd_abs(actual-ref)));n+=3;}
printf("positive-height retained256x128 finite-SKY CPU field: filteredRGB_RMSE=%.6f directRGB_RMSE=%.6f resamplingRGB_RMSE=%.6f maxChannel=%.6f samples=%u\n",std::sqrt(filtered/n),std::sqrt(direct/n),std::sqrt(resampling/n),maxerr,n);}
