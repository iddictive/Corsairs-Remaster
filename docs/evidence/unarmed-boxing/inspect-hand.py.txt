from PIL import Image
s=json.load(open('src/assets/roundhouse/rigs.json'))['rigs']['man'];r=source_rig(s)
verts,faces=animated_mesh(Path('docs/evidence/roundhouse-kick/inputs/Blad_0.gm').read_bytes());bound=bind_skin(s,verts)

for j,p in enumerate(r['parents']):
 if p in desc:desc.add(j)

faces=[f for f in faces if all(mask[i] for i in f)]
canvas=Image.new('RGB',(1800,500));bank=rig(Path('experiments/native-metal/.cache/unarmed/RESOURCE/animation/Unarmed_man.an').read_bytes());m=json.load(open('src/assets/unarmed/manifest.json'));f=m['banks']['man']['actions']['unarmed_uppercut'][0]+13

 for col,angle in enumerate((0,45,90)):
  a=math.radians(angle);right=np.array((math.cos(a),0,-math.sin(a)));xx=selected@right;lo,hi=xx.min(),xx.max();floor=selected[:,1].min()-.01;zoom=min(560/(hi-lo),210/(selected[:,1].max()-floor))
  canvas.paste(v.panel(points,faces,angle,(600,250),f'{name} {angle}',((lo+hi)/2,floor,zoom)),(col*600,row*250))
